            if (err) console.error(' នប កង របិទឯក រ:', err); 
        }); 
    }); 
});

const fs = require('fs'); 
const path = require('path'); 
 
const filename = 'output.txt'; 
const filePath = path.join(__dirname, filename); // Creates a path in the current directory 
 
// 1. Data to be written, including data before and after the part we want to write 
const dataToWrite = '---HEADER---Data to be written---FOOTER---'; 
 
// 2. Create the Buffer from the string data 
// The full buffer will contain the binary representation of the entire string 
const buffer = Buffer.from(dataToWrite, 'utf8'); 
 
// Parameters for fs.write (fd, buffer, offset, length, position, callback) 
const bufferOffset = 12; // Start reading from the 13th byte (index 12) of the buffer ("D" 
of "Data...") 
const length = 18;       // Write 18 bytes from the buffer (covers "Data to be written") 
const filePosition = 0;  // Start writing at the beginning of the file (0) 
 
// 3. Open the file for writing (flag 'w') 
fs.open(filePath, 'w', (err, fd) => { 
    if (err) { 
        console.error('Error opening file:', err); 
        return; 
    } 
 
    // 4. Use fs.write to write the part of the buffer 
    fs.write(fd, buffer, bufferOffset, length, filePosition, (err, writtenBytes, buffer) => 
{ 
        if (err) { 
            console.error('Error writing to file:', err); 
        } else { 
            console.log(`Successfully wrote ${writtenBytes} bytes to ${filename}.`); 
            console.log(`Content written: "${buffer.toString('utf8', bufferOffset, 
bufferOffset + writtenBytes)}"`); 
        } 
 
        // 5. Close the file descriptor (important!) 
        fs.close(fd, (err) => { 
            if (err) { 
                console.error('Error closing file:', err); 
            } else { 
                console.log('File closed successfully.'); 
            } 
        }); 
    }); 
});

// Check File Existence
const fs = require('fs').promises; // េបកំែណ Promise របស់ fs 
 
async function checkFileExistence(filePath) { 
    try { 
        // fs.constants.F_OK ន់ែតពិនិតេមល ឯក រ នឬអត់ 
        await fs.access(filePath, fs.constants.F_OK); 
        console.log(`✅ ឯក រ "${filePath}" ន។`); 
        return true; 
    } catch (error) { 
        // បសិនេប នកំហុស នន័យ ឯក រមិន ន ឬមិន ចចូលេប ន។ 
        if (error.code === 'ENOENT') { 
            console.log(`❌ ឯក រ "${filePath}" មិន នេទ។`); 
        } else { 
            console.log(`⚠ កំហុសកង រចូលេបឯក រ៖ ${error.message}`); 
        } 
        return false; 
    } 
} 
 
// ឧ ហរណ៍ៃន រេប ស់ 
checkFileExistence('./my-file.txt'); 
checkFileExistence('./non-existent-file.txt');

            } 
            return; // េចញពី Callback 
        } 
 
        // កូដេនះដំេណរ រែតេ េពលេ គជ័យ 
        console.log(`✅ ឯក រ "${filePath}" ន។`); 
    });

}

    } 
}

// បេងតម៉ូឌុលែដល return EventEmitter 
function createCounter() { 
    const emitter = new EventEmitter(); 
    let count = 0; 
 
    setInterval(() => { 
        count++; 
        emitter.emit('count', count); // ប នពឹតិ រណ៍ 'count' 
    }, 1000); 
 
    return emitter; 
}

    console.log(`ចំនួនបចបន: ${value}`); 
});

const express = require('express'); 
const connectDB = require('./db'); 
const User = require('./models/User'); 
require('dotenv').config(); 
 
const app = express(); 
const port = process.env.PORT || 3000; 
 
connectDB(); 
app.use(express.json()); 
 
// Route ស ប់បេងត user ថី 
app.post('/users', async (req, res) => { 
    try { 
        const user = new User(req.body); 
        await user.save(); 
        res.status(201).send(user); 
    } catch (error) { 
        res.status(400).send(error.message); 
    } 
}); 
 
// Route ស ប់ ប់េដត user 
app.put('/users/:id', async (req, res) => { 
    try { 
        const user = await User.findByIdAndUpdate( 
            req.params.id, 
            req.body, 
            { new: true, runValidators: true } 
        ); 
        if (!user) { 
            return res.status(404).send('រក user មិនេឃញ'); 
        } 
        res.send(user); 
    } catch (error) { 
        res.status(400).send(error.message); 
    } 
}); 
 
app.listen(port, () => { 
    console.log(`Server កំពុងដំេណរ រេ http://localhost:${port}`); 
});
