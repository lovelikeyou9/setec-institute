-- Using Function ADD_MONTHS() calculator => Month, Year
-- 1. GET CURRENT DATE, CURRENT DATE + 2 MONTHS
-- CURRENT DATE - 1 MONTH
SELECT SYSDATE, ADD_MONTHS(SYSDATE,2) AS "+ Months"
      ,ADD_MONTHS(SYSDATE,-1) AS "-1 Month"
FROM DUAL;

-- 2. GET CURRENT DATE, CURRENT DATE + 2 Days + 3 Months - 2 Years
-- 1 YEAR = 12 Months
SELECT SYSDATE, ADD_MONTHS(SYSDATE,2 * 12) AS "-1 Year"
FROM DUAL;

-- 3. GET CURRENT DATE, CURRENT DATE + 2 Days + 3 Months - 2 Years
-- 4. CALCULATOR BETWEEN of MONTHS & YEARS
SELECT SYSDATE, HIRE_DATE
      ,ROUND(MONTHS_BETWEEN(SYSDATE,HIRE_DATE)) AS "Month of Working"
      ,ROUND(MONTHS_BETWEEN(SYSDATE,HIRE_DATE)/12) AS "Year of Working"
FROM EMPLOYEES;
