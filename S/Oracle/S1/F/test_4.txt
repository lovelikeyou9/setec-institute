
-- COUNT OF letter 'e'
SELECT 'WelcometoSETECSV2.6',
       REGEXP_COUNT(LOWER('WelcometoSETECSV2.6'),'e') AS COUNT_E
FROM DUAL;

-- COUNT of letter 'e'
SELECT FIRST_NAME, REGEXP_COUNT(LOWER(FIRST_NAME),'e') FROM EMPLOYEES
WHERE REGEXP_COUNT(LOWER(FIRST_NAME),'e')<>0;

-- Using LPAD & RPAD function => Format ID => 001
SELECT ROWNUM, LPAD(ROWNUM,3), FIRST_NAME, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES;

SELECT ROWNUM, LPAD(ROWNUM,4,0), FIRST_NAME, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES;

SELECT LAST_NAME, SALARY, RPAD(SALARY,2,0) FROM EMPLOYEES;

SELECT LAST_NAME, SALARY, RPAD(SALARY,5,0) FROM EMPLOYEES;

-- Using Replace() Function
SELECT LAST_NAME, JOB_ID FROM EMPLOYEES;

SELECT LAST_NAME, JOB_ID, REPLACE(JOB_ID,'_','+') FROM EMPLOYEES;

-- Using Nested Function
SELECT LAST_NAME, JOB_ID, REPLACE(REPLACE(JOB_ID,'_','-'),'CLERK','STAFF') FROM EMPLOYEES;

SELECT SALARY, LPAD(SALARY,5,'*') FROM EMPLOYEES;

SELECT SALARY, RPAD(LPAD(SALARY,5,'*'),7,'#') FROM EMPLOYEES;

-- Using TRIM() Function
SELECT 'HelloValloH', TRIM('H' FROM 'HelloValloH') AS TRIM_H FROM EMPLOYEES;

SELECT 'HelloValloH', TRIM('H' FROM 'HelloValloH') AS TRIM_H
      ,EMAIL, TRIM('S' FROM EMAIL) AS TRIM_EMAIL
FROM EMPLOYEES;
