-- Function TO_DATE() using in where condition
SELECT EMPLOYEE_ID, LAST_NAME, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE='03-JAN-90'; -- DD-MON-RR

-- HIRE_DATE => 03-01-1990 => DD-MM-YYYY
SELECT EMPLOYEE_ID, LAST_NAME, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE=TO_DATE('03-01-1990','DD-MM-YYYY');

SELECT EMPLOYEE_ID, LAST_NAME, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE=TO_DATE('17-MAY-2025','DD-MONTH-YYYY');

-- Function TO_CHAR() using Format on Columns
SELECT LAST_NAME, HIRE_DATE, TO_CHAR(HIRE_DATE,'DD-MM-YYYY') AS "Hire"
            --> Format HIRE_DATE: DD-MM-YYYY
FROM EMPLOYEES;

--> Format: HIRE_DATE: DY DD-MONTH-YEAR
SELECT LAST_NAME, HIRE_DATE, TO_CHAR(HIRE_DATE,'DY DD-MONTH-YEAR') AS DYMONYEAR FROM EMPLOYEES;

SELECT LAST_NAME, HIRE_DATE, TO_CHAR(HIRE_DATE,'FMDY DD-MONTH-YEAR') AS DYMONYEAR FROM EMPLOYEES;

-- GET CURRENT DATE & TIME -> Format: DD-MM-YYYY 07:55:40 AM => HH:MI:SS AP/PM
SELECT SYSDATE FROM DUAL;
SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY') FROM DUAL;
SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH') FROM DUAL;
SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH:MI') FROM DUAL;
SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH:MI:SS') FROM DUAL;
SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH:MI:SS AM') AS "DATE&TIME" FROM DUAL;
-- 24h => HH24:MI:SS AM/PM
SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH24:MI:SS AM') AS "DATE&TIME" FROM DUAL;

SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH24:MI:SS AM') AS "DATE&TIME"
      ,TO_CHAR(SYSDATE,'DAY DD-MM-YYYY') AS Translate
FROM DUAL; -- Of Date

SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH24:MI:SS AM') AS "DATE&TIME"
      ,TO_CHAR(SYSDATE,'DAY "Of Date" DD-MM-YYYY') AS Translate
FROM DUAL; -- Of Date

SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH24:MI:SS AM') AS "DATE&TIME"
      ,TO_CHAR(SYSDATE,'DAY "Of Date" DD-MM-YYYY YYYY') AS Translate
FROM DUAL; -- Of Date

SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH24:MI:SS AM') AS "DATE&TIME"
      ,TO_CHAR(SYSDATE,'DAY "Of Date" DD-MM-YYYY YYYYspth') AS Translate
FROM DUAL; -- Of Date

SELECT SYSDATE, TO_CHAR(SYSDATE,'DD-MM-YYYY HH24:MI:SS AM') AS "DATE&TIME"
      ,TO_CHAR(SYSDATE,'DAY "Of Date" DD-MM-YYYY "to" YYYYspth') AS Translate
FROM DUAL; -- Of Date
