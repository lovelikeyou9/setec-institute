SELECT LOWER(first_name) AS LFIRST, UPPER(last_name) AS ULAST FROM EMPLOYEES;

SELECT FIRST_NAME, LAST_NAME, INITCAP(FIRST_NAME) || ' ' || INITCAP(LAST_NAME) AS "Fullname", 
LOWER(FIRST_NAME) || LOWER(LAST_NAME) || 'setec@gmail.com' AS "Email"
FROM EMPLOYEES;

-- with where condition
SELECT * FROM DEPARTMENTS
WHERE LOWER(DEPARTMENT_NAME)=LOWER('it');

--with concat function
SELECT FIRST_NAME, LAST_NAME, CONCAT(CONCAT(FIRST_NAME,' '),LAST_NAME) AS "Fullname" FROM EMPLOYEES;
