-- 1
SELECT last_name, salary,
    INSTR(last_name, 'a') AS pos_a
FROM employees
WHERE INSTR(last_name, 'a') >= 3;

-- 2
SELECT
    last_name,
    LENGTH(last_name) AS len_last
FROM employees
WHERE LENGTH(last_name) >= 5;

-- 3
SELECT
    first_name,
    LENGTH(first_name) AS len_first
FROM employees
WHERE LENGTH(first_name) <= 5;

-- 4
SELECT
    last_name,
    INSTR(last_name, ' ') AS pos_space
FROM employees
WHERE INSTR(last_name, ' ') > 0;

-- 5
SELECT
    last_name, salary,
    INSTR(last_name, 'r') AS pos_r
FROM employees
WHERE INSTR(last_name, 'r') >= 5;

-- 6
SELECT
    last_name,
    LENGTH(last_name) AS len_last
FROM employees
WHERE LENGTH(last_name) <= 6;

-- 7
SELECT
    last_name,
    SUBSTR(last_name, 1, 1) AS first_char
FROM employees
WHERE SUBSTR(last_name, 1, 1) IN ('A','B','C','D','E','F');

-- 8
SELECT
    last_name,
    SUBSTR(last_name, -2) AS last_2chars
FROM employees
WHERE SUBSTR(last_name, -2) IN ('as','ng','ey','ns');

-- 9
SELECT
    last_name,
    SUBSTR(last_name, 1, 4) AS first_4chars,
    SUBSTR(last_name, -4) AS last_4chars
FROM employees;
