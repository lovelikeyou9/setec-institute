-- USING ROUND(), TRUNC(), MOD() FUNCTION
SELECT 340.927
      ,ROUND(340.927,2)
      ,ROUND(340.927)
      ,ROUND(340.927,0)
      ,TRUNC(340.927,2)
      ,TRUNC(340.927)
      ,TRUNC(340.927,0)
      ,MOD(100,50)
      ,MOD(340.927,50)
FROM DUAL;

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES;

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)=0;

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)<>0;

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)=0
ORDER BY SALARY DESC
FETCH FIRST 4 ROWS ONLY

SELECT EMPLOYEE_ID, LAST_NAME, ROUND(SALARY/12), JOB_ID FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)<>0;

-- Format currency: $12,000.00 => Function TO_CHAR($,0,9)
-- Delete space => FM
SELECT EMPLOYEE_ID, LAST_NAME, TO_CHAR(ROUND(SALARY/12),'$99,999,999.00'), JOB_ID FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)<>0;

SELECT EMPLOYEE_ID, LAST_NAME, TO_CHAR(ROUND(SALARY/12),'FM$99,999,999.00') AS SAL_MONTHLY, JOB_ID FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)<>0;

SELECT EMPLOYEE_ID, LAST_NAME
      ,TO_CHAR(ROUND(SALARY/12),'FM$99,999,999.00') AS SAL_Dollar
      ,'€' || TO_CHAR(ROUND(SALARY/12),'FM99,999,999.00') AS SAL_Euro
      ,JOB_ID
FROM EMPLOYEES
WHERE MOD(EMPLOYEE_ID,2)<>0;
