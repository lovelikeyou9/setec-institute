--  1
SELECT employee_id, last_name, salary
FROM employees
WHERE salary IN (2500, 3100, 3500);

--  2
SELECT department_id, department_name, location_id
FROM departments
WHERE location_id <> 1700;

--  3
SELECT department_id, department_name, manager_id
FROM departments
WHERE manager_id IN (100, 200, 205);

--  4
SELECT employee_id, last_name, salary, job_id
FROM employees
WHERE job_id IN ('IT_PROG', 'ST_MAN', 'AD_VP');

--  5
SELECT last_name, salary, hire_date
FROM employees
WHERE hire_date IN (TO_DATE('07-JUN-94','DD-MON-RR'),
                    TO_DATE('21-MAY-91','DD-MON-RR'),
                    TO_DATE('16-NOV-99','DD-MON-RR'));
--  or
SELECT last_name, salary, hire_date
FROM employees
WHERE hire_date IN ('07-JUN-94','21-MAY-91','16-NOV-99');

--  6
SELECT last_name, salary, job_id
FROM employees
WHERE job_id <> 'IT_PROG';

--  7
SELECT employee_id, first_name, salary
FROM employees
WHERE salary BETWEEN 2600 AND 3200;

--  8
SELECT employee_id, last_name, hire_date
FROM employees
WHERE hire_date BETWEEN TO_DATE('01-JAN-95','DD-MON-RR') 
                    AND TO_DATE('30-DEC-98','DD-MON-RR');
--  or
SELECT employee_id, last_name, hire_date
FROM employees
WHERE hire_date BETWEEN '01-JAN-95' AND '30-DEC-98';

--  9
SELECT first_name || ' ' || last_name AS FullName,
       12 * salary + 1200 AS Total_Salary,
       hire_date
FROM employees
WHERE hire_date < TO_DATE('11-NOV-1990','DD-MON-YYYY');
--  or
SELECT first_name || ' ' || last_name AS FullName,
       12 * salary + 1200 AS Total_Salary,
       hire_date
FROM employees
WHERE hire_date < TO_DATE('11-11-1990','DD-MM-YYYY');
