SELECT EMPLOYEE_ID, SALARY, JOB_ID, DEPARTMENT_ID FROM EMPLOYEES
WHERE DEPARTMENT_ID=80;

-- Who have salary equal to(=) 17000
SELECT EMPLOYEE_ID, LAST_NAME, SALARY FROM EMPLOYEES
WHERE SALARY=17000;

-- Who have job_id is or equal to(=) IT_PROG
SELECT * FROM DEPARTMENTS
WHERE DEPARTMENT_NAME='Sales';

SELECT LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES
WHERE JOB_ID='IT_PROG';

-- Default Date format in Oracle: DD-MON-RR
-- Who started working on date: 07-JUNE-1994
SELECT LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES 
WHERE HIRE_DATE='07-JUN-94'; -- 07-JUNE-1994

-- Who started working on date: 07-06-1994
SELECT LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE='07-06-1994'; -- DD-MM-YYYY  -- ERROR

SELECT LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE=TO_DATE'07-06-1994'; -- DD-MM-YYYY  -- ERROR

SELECT LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE=TO_DATE('07-06-1994','DD-MM-YYYY'); -- DD-MM-YYYY  -- WORK

-- Who started working before date: 01-JAN-89
SELECT LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE < '01-JAN-89';

-- Who started working after date: 30-DEC-98
SELECT LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE > '30-DEC-98';
