SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID FROM EMPLOYEES
WHERE DEPARTMENT_ID <> 50;

SELECT LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES
WHERE JOB_ID != 'ST_CLERK';

-- 07-JUN-94
SELECT LAST_NAME, SALARY, JOB_ID, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE ^= '07-JUN-94';

-- Operator BETWEEN ... AND ...
-- Who have salary range 9000 to 11000
SELECT EMPLOYEE_ID, LAST_NAME, JOB_ID, SALARY FROM EMPLOYEES
WHERE SALARY BETWEEN 9000 AND 11000;

-- Who started working 01-JAN-98 to 30-DEC-00
SELECT EMPLOYEE_ID, LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE BETWEEN '01-JAN-98' AND '30-DEC-00';

-- Operator IN (..., ..., ....)
SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID FROM EMPLOYEES
WHERE DEPARTMENT_ID IN(10);

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID FROM EMPLOYEES
WHERE DEPARTMENT_ID IN(10,20);

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID FROM EMPLOYEES
WHERE DEPARTMENT_ID IN(10,20,80);

-- Who working in department_id such as 10,20 and 80
SELECT EMPLOYEE_ID, LAST_NAME, SALARY, JOB_ID FROM EMPLOYEES
WHERE JOB_ID IN ('AD_VP','SA_REP','IT_PROG');

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, HIRE_DATE FROM EMPLOYEES
WHERE HIRE_DATE IN ('03-JAN-90','07-JUN-94','11-MAY-25');
