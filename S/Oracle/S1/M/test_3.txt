SELECT ROWNUM, LAST_NAME, SALARY,
       12*SALARY+100,
       12*(SALARY+100)

SELECT LAST_NAME,
       SALARY,
       NVL(COMMISSION_PCT,0),
       SALARY * NVL(COMMISSION_PCT,0) AS Commission_Amount
FROM EMPLOYEES;

SELECT DEPARTMENT_NAME AS name,
       MANAGER_ID AS mgride
FROM DEPARTMENTS;

SELECT LAST_NAME,
       12*(SALARY*300) AS Total_Salary
FROM EMPLOYEES;

SELECT LAST_NAME AS "name",
       12*SALARY+300 AS "Total Salary",
       DEPARTMENT_ID AS "DEPT #@%"
FROM EMPLOYEES;
