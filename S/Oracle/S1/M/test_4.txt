SELECT DEPARTMENT_NAME, MANAGER_ID,
       DEPARTMENT_NAME || q'[ Department's Manager ID: ]'
       || MANAGER_ID
       AS "Name and Manager"
FROM DEPARTMENTS;

SELECT LAST_NAME, SALARY, LAST_NAME || ' has total salary: ' || SALARY
       AS "Name and Salary",
       LAST_NAME || q'{'s has total salary's: }' || SALARY AS "Total Salary"
FROM EMPLOYEES;
