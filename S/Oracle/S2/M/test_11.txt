DROP TABLE PRODUCTS;

-- Create Table Category
CREATE TABLE CATEGORY(
  category_id NUMBER(10) PRIMARY KEY,
  category_name VARCHAR2(30) NOT NULL UNIQUE,
  description VARCHAR2(30),
  created_date DATE DEFAULT SYSDATE
);

SELECT * FROM CATEGORY;

-- Create Table PRODUCTS
CREATE TABLE PRODUCTS(
  product_id NUMBER(10) PRIMARY KEY,
  product_name VARCHAR2(30) UNIQUE NOT NULL,
  price NUMBER(20, 2) NOT NULL CHECK(PRICE>0),
  qty NUMBER(20) DEFAULT 0,
  amount NUMBER(20, 2) GENERATED ALWAYS AS(PRICE * QTY),
  description VARCHAR2(30),
  created_date DATE DEFAULT SYSDATE,
  cat_id NUMBER(10),
  CONSTRAINT FK_PRODUCTS_CATEGORY FOREIGN KEY(cat_id) REFERENCES CATEGORY(CATEGORY_ID)
  ON DELETE CASCADE
);

SELECT * FROM PRODUCTS;
