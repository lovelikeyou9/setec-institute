CREATE TABLE PRODUCTS(
  PRODUCT_ID NUMBER(10) GENERATED AS IDENTITY PRIMARY KEY,
  PRODUCT_NAME VARCHAR2(30) NOT NULL UNIQUE,
  QTY NUMBER(30) NOT NULL CHECK(QTY>=0),
  PRICE NUMBER(20,2) NOT NULL CHECK(PRICE BETWEEN 0 AND 200),
  AMOUNT NUMBER(20,2) GENERATED ALWAYS AS(QTY * PRICE),
  PRODUCT_SIZE VARCHAR2(10) CHECK(PRODUCT_SIZE IN('M','S','L','XL')),
  DESCRITION VARCHAR2(50),
  CREATE_DATE DATE DEFAULT TO_DATE('12-04-2026','DD-MM-YYYY') -- DD-MM-YYYY
);
