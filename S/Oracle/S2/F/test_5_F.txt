-- Using GROUP BY => FUNCTION(MAX, MIN, AVG, SUM)
CREATE OR REPLACE VIEW VIEW_GROUPBY_HAVING AS
SELECT DEPARTMENT_ID, 
  TO_CHAR(SUM(SALARY), 'fm$99,999,999.00') AS "Total Salary",
  COUNT(*) AS "COUNT OF DEPT"
FROM EMPLOYEES
WHERE DEPARTMENT_ID IS NOT NULL
HAVING COUNT(*)>=3 AND SUM(SALARY)>20000
GROUP BY DEPARTMENT_ID
ORDER BY "Total Salary" DESC;

CREATE OR REPLACE VIEW VIEW_GROUPBY_HAVING AS
SELECT DEPARTMENT_ID, 
  TO_CHAR(SUM(SALARY), 'fm$99,999,999.00') AS "Total Salary",
  MIN(SALARY) AS "Minimum Salary",
  MAX(SALARY) AS "Maximum Salary",
  COUNT(*) AS "COUNT OF DEPT"
FROM EMPLOYEES
WHERE DEPARTMENT_ID IS NOT NULL
HAVING COUNT(*)>=3 AND SUM(SALARY)>20000
GROUP BY DEPARTMENT_ID
ORDER BY "Total Salary" DESC;

-- Display view named VIEW_GROUPBY_HAVING
SELECT * FROM VIEW_GROUPBY_HAVING;
