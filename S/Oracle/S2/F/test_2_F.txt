-- 1. CREATE VIEW NAME USING ORACLE
SELECT * FROM EMPLOYEES;

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID
FROM EMPLOYEES;

SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID
FROM EMPLOYEES WHERE SALARY >= 9000;

CREATE OR REPLACE VIEW VIEW_EMPSV26 AS
SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID
FROM EMPLOYEES WHERE SALARY >= 9000;

-- 2. Display view named VIEW_EMPSV26
SELECT * FROM VIEW_EMPSV26;

CREATE OR REPLACE VIEW VIEW_EMPSV26 AS
SELECT EMPLOYEE_ID, LAST_NAME, SALARY, DEPARTMENT_ID
FROM EMPLOYEES WHERE SALARY >= 9000 AND DEPARTMENT_ID=90;

SELECT * FROM VIEW_EMPSV26;

-- 3. CREATE VIEW NAME USING FORMAT COLUMN
SELECT * FROM EMPLOYEES;

SELECT FIRST_NAME, LAST_NAME, SALARY, HIRE_DATE
FROM EMPLOYEES;

SELECT ROWNUM AS "EMP #", FIRST_NAME, LAST_NAME, SALARY, HIRE_DATE
FROM EMPLOYEES;

SELECT ROWNUM AS "EMP #", UPPER(FIRST_NAME) AS FIRST_NAME, 
  LOWER(LAST_NAME) AS LAST_NAME, 
  FIRST_NAME || ' ' || LAST_NAME AS "Full Name",
  LOWER(FIRST_NAME || LAST_NAME || '168@gmail.com') AS "Email",
  TO_CHAR(SALARY,'fm$99,999,999.00') AS CURRENCY, 
  TO_CHAR(HIRE_DATE,'dd-mm-yyyy') AS HIRE_DATE
FROM EMPLOYEES;

CREATE OR REPLACE VIEW VIEW_EMPCOLUMNFRM AS
SELECT ROWNUM AS "EMP #", UPPER(FIRST_NAME) AS FIRST_NAME, 
  LOWER(LAST_NAME) AS LAST_NAME, 
  FIRST_NAME || ' ' || LAST_NAME AS "Full Name",
  LOWER(FIRST_NAME || LAST_NAME || '168@gmail.com') AS "Email",
  TO_CHAR(SALARY,'fm$99,999,999.00') AS CURRENCY, 
  TO_CHAR(HIRE_DATE,'dd-mm-yyyy') AS HIRE_DATE
FROM EMPLOYEES;

-- 4. Display view named VIEW_EMPCOLUMNFRM
SELECT * FROM VIEW_EMPCOLUMNFRM;
