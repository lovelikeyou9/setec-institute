  d.DEPARTMENT_NAME

ON e.DEPARTMENT_ID = d.DEPARTMENT_ID;

SELECT e.EMPLOYEE_ID, e.LAST_NAME, e.SALARY, e.DEPARTMENT_ID,
  d.DEPARTMENT_NAME
FROM  EMPLOYEES e JOIN DEPARTMENTS d
ON e.DEPARTMENT_ID = d.DEPARTMENT_ID
WHERE d.DEPARTMENT_NAME NOT IN ('Shipping','Sales');

SELECT e.EMPLOYEE_ID, e.LAST_NAME, e.SALARY, e.DEPARTMENT_ID,
  d.DEPARTMENT_NAME
FROM  EMPLOYEES e JOIN DEPARTMENTS d
ON e.DEPARTMENT_ID = d.DEPARTMENT_ID
WHERE d.DEPARTMENT_NAME NOT IN ('Shipping','Sales', 'IT');

SELECT e.EMPLOYEE_ID, e.LAST_NAME, e.SALARY, e.DEPARTMENT_ID,
  d.DEPARTMENT_NAME
FROM  EMPLOYEES e JOIN DEPARTMENTS d
ON e.DEPARTMENT_ID = d.DEPARTMENT_ID
WHERE d.DEPARTMENT_NAME NOT IN ('Shipping','Sales', 'IT')
  AND e.SALARY > 9000 ORDER BY e.SALARY DESC;

CREATE OR REPLACE VIEW VIEW_EMP_DEPT AS
SELECT e.EMPLOYEE_ID, e.LAST_NAME, e.SALARY, e.DEPARTMENT_ID,
  d.DEPARTMENT_NAME
FROM  EMPLOYEES e JOIN DEPARTMENTS d
ON e.DEPARTMENT_ID = d.DEPARTMENT_ID
WHERE d.DEPARTMENT_NAME NOT IN ('Shipping','Sales', 'IT')
  AND e.SALARY > 9000 ORDER BY e.SALARY DESC;

-- Display view named VIEW EMP_DEPT
SELECT * FROM VIEW_EMP_DEPT;
