-- 1. Create table named STUDENTS

  STUDENT_ID NUMBER(10),
  STUDENT_NAME VARCHAR2(45)
);

SELECT * FROM STUDENTS;

-- 2. add columns gender varchar(10)

SELECT * FROM STUDENTS;

-- 3. Add Mutiple colums age and subject
ALTER TABLE STUDENTS ADD(
  AGE NUMBER (10) NOT NULL CHECK(AGE>0),
  SUBJECT VARCHAR2(50) NOT NULL
);

SELECT * FROM STUDENTS;

-- 4. Remove column age from STUDENTS
ALTER TABLE STUDENTS DROP COLUMN AGE;

SELECT * FROM STUDENTS;

-- 5. Remove multiple columns gender & subject
ALTER TABLE STUDENTS DROP( GENDER, SUBJECT );
