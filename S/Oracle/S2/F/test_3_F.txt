-- 1. CREATE VIEW USING NESTED FUNCTION
SELECT EMPLOYEE_ID,
  FIRST_NAME, LAST_NAME,
  SALARY*4000 AS RIEL, JOB_ID
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS;

SELECT EMPLOYEE_ID,
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  SALARY*4000 AS RIEL, JOB_ID
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

SELECT EMPLOYEE_ID,
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  JOB_ID
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

SELECT RPAD(LPAD(EMPLOYEE_ID, 5, '*'), 8, '#') AS "EMP #",
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  JOB_ID
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

SELECT RPAD(LPAD(EMPLOYEE_ID, 5, '*'), 8, '#') AS "EMP #",
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  REPLACE(JOB_ID, '_', '-')
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

SELECT RPAD(LPAD(EMPLOYEE_ID, 5, '*'), 8, '#') AS "EMP #",
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  REPLACE(REPLACE(JOB_ID, '_', '-'), 'REP', 'SELL') AS LEVEL_JOB
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

SELECT RPAD(LPAD(EMPLOYEE_ID, 5, '*'), 8, '#') AS "EMP #",
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  REPLACE(REPLACE(JOB_ID, '_', '-'), 'REP', 'SELL') AS LEVEL_JOB,
  NVL(TO_CHAR(DEPARTMENT_ID), 'No Department') AS Department
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

SELECT RPAD(LPAD(EMPLOYEE_ID, 5, '*'), 8, '#') AS "EMP #",
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  REPLACE(REPLACE(JOB_ID, '_', '-'), 'REP', 'SELL') AS LEVEL_JOB,
  NVL(TO_CHAR(DEPARTMENT_ID), 'No Department') AS Department,
  NVL2(COMMISSION_PCT, 'Have Commission', 'No Commission') AS "Commission"
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;

CREATE OR REPLACE VIEW VIEW_NESTEDFUNC AS
SELECT RPAD(LPAD(EMPLOYEE_ID, 5, '*'), 8, '#') AS "EMP #",
  UPPER(FIRST_NAME || ' ' || LAST_NAME) AS "FullName",
  'KHR' || TO_CHAR(SALARY*4000, 'fm99,999,999.00') AS RIEL, 
  REPLACE(REPLACE(JOB_ID, '_', '-'), 'REP', 'SELL') AS LEVEL_JOB,
  NVL(TO_CHAR(DEPARTMENT_ID), 'No Department') AS Department,
  NVL2(COMMISSION_PCT, 'Have Commission', 'No Commission') AS "Commission"
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY RIEL DESC
OFFSET 4 ROWS FETCH NEXT 5 ROWS ONLY;
