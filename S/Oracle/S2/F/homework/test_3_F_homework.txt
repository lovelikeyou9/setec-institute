CREATE OR REPLACE VIEW view_compare AS
SELECT  FIRST_NAME, LAST_NAME,
  LENGTH(FIRST_NAME) AS Count_Last,
  LENGTH(LAST_NAME) AS Count_First,

  NULLIF(LENGTH(FIRST_NAME), LENGTH(LAST_NAME)) AS Compare,

  NVL2(
    NULLIF(LENGTH(FIRST_NAME), LENGTH(LAST_NAME)), 'Different', 'The Same'
  ) AS Result

FROM EMPLOYEES;

SELECT * FROM VIEW_COMPARE;

-- 2
CREATE OR REPLACE VIEW view_workingdate AS
SELECT LAST_NAME, SYSDATE AS Current_Date, HIRE_DATE,
  ROUND(SYSDATE - HIRE_DATE, 0) AS Total_day,
  ROUND((SYSDATE - HIRE_DATE) / 30, 0) AS Total_Month,
  ROUND((SYSDATE - HIRE_DATE) / 365, 0) AS Total_Year
FROM EMPLOYEES;

SELECT * FROM view_workingdate;
