-- 1. VIEW_COMPLEXEMP01
SELECT SALARY FROM EMPLOYEES;

CREATE OR REPLACE VIEW VIEW_COMPLEXEMP01 AS
SELECT 
    TO_CHAR(SUM(salary), '$99,999,999.00') AS "Total Salary",
    TO_CHAR(MIN(salary), '$99,999,999.00') AS "Minimum Salary",
    TO_CHAR(MAX(salary), '$99,999,999.00') AS "Maximum Salary"
FROM employees
WHERE department_id = 60;

SELECT * FROM VIEW_COMPLEXEMP01;

-- 2. VIEW_COMPLEXEMP02
CREATE OR REPLACE VIEW VIEW_COMPLEXEMP02 AS
SELECT
    TO_CHAR(
        MAX(CASE WHEN department_id = 50 THEN salary END),
        '$99,999,999.00'
    ) AS MAX_DEPT_50_SALARY,

    TO_CHAR(
        MAX(DECODE(department_id, 80, salary)),
        '99,999,999.00$'
    ) AS MAX_DEPT_80_SALARY
FROM employees;

SELECT * FROM VIEW_COMPLEXEMP02;

-- 3. VIEW_COMPLEXEMP03
CREATE OR REPLACE VIEW VIEW_COMPLEXEMP03 AS
SELECT
    COUNT(*) AS Total_Employees,
    MIN(hire_date) AS EARLIEST_HIRE_DATE,
    MAX(hire_date) AS LATEST_HIRE_DATE,

    ROUND(AVG(MONTHS_BETWEEN(SYSDATE, hire_date) / 12)) AS AVG_YEARS_EMPLOYED,

    ROUND(MIN(MONTHS_BETWEEN(SYSDATE, hire_date) / 12)) AS MIN_YEARS_EMPLOYED,

    ROUND(MAX(MONTHS_BETWEEN(SYSDATE, hire_date) / 12)) AS MAX_YEARS_EMPLOYED
FROM employees;

SELECT * FROM VIEW_COMPLEXEMP03;

-- 4. VIEW_COMPLEXEMP04
CREATE OR REPLACE VIEW VIEW_COMPLEXEMP04 AS
SELECT
    MIN(NVL(commission_pct, 0)) AS MIN_COMMISSION_PCT,
    ROUND(AVG(NVL(commission_pct, 0)), 0) AS AVG_COMMISSION_PCT
FROM employees;

SELECT * FROM VIEW_COMPLEXEMP04;
