CREATE OR REPLACE VIEW View_EMPO AS
SELECT EMPLOYEE_ID,
  FIRST_NAME || ' ' || LAST_NAME AS FULLNAME,
  SALARY
FROM EMPLOYEES WHERE SALARY > 8000;

CREATE OR REPLACE VIEW View_EMPO AS
SELECT EMPLOYEE_ID,
  FIRST_NAME || ' ' || LAST_NAME AS FULLNAME,
  SALARY
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 0
ORDER BY SALARY DESC
FETCH FIRST 7 ROWS ONLY;  -- Correct

SELECT * FROM View_EMPO;

CREATE OR REPLACE VIEW View_EMP1 AS
SELECT  EMPLOYEE_ID,
  FIRST_NAME || ' ' || LAST_NAME AS FULLNAME,
  SALARY
FROM EMPLOYEES WHERE SALARY BETWEEN 2000 AND 8000;

CREATE OR REPLACE VIEW View_EMP1 AS
SELECT  EMPLOYEE_ID,
  FIRST_NAME || ' ' || LAST_NAME AS FULLNAME,
  SALARY
FROM EMPLOYEES WHERE MOD(EMPLOYEE_ID, 2) = 1
ORDER BY SALARY ASC
FETCH FIRST 7 ROWS ONLY;  -- correct

SELECT * FROM View_EMP1;
