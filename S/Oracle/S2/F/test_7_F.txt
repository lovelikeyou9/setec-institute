SELECT d.DEPARTMENT_ID, d.DEPARTMENT_NAME, d.LOCATION_ID, l.CITY
FROM DEPARTMENTS d JOIN LOCATIONS l
ON d.LOCATION_ID = l.LOCATION_ID

SELECT d.DEPARTMENT_ID, d.DEPARTMENT_NAME, d.LOCATION_ID, l.CITY,
  l.COUNTRY_ID , c.COUNTRY_NAME
FROM DEPARTMENTS d JOIN LOCATIONS l
ON d.LOCATION_ID = l.LOCATION_ID
JOIN COUNTRIES c
ON l.COUNTRY_ID = c.COUNTRY_ID;

SELECT d.DEPARTMENT_ID, d.DEPARTMENT_NAME, d.LOCATION_ID, l.CITY,
  l.COUNTRY_ID , c.COUNTRY_NAME
FROM DEPARTMENTS d JOIN LOCATIONS l
ON d.LOCATION_ID = l.LOCATION_ID
JOIN COUNTRIES c
ON l.COUNTRY_ID = c.COUNTRY_ID
WHERE l.CITY <> 'Seattle' ORDER BY d.LOCATION_ID DESC;

CREATE OR REPLACE VIEW VIEW_DEPT_LOCATE_COUNTRY AS
SELECT d.DEPARTMENT_ID, d.DEPARTMENT_NAME, d.LOCATION_ID, l.CITY,
  l.COUNTRY_ID , c.COUNTRY_NAME
FROM DEPARTMENTS d JOIN LOCATIONS l
ON d.LOCATION_ID = l.LOCATION_ID
JOIN COUNTRIES c
ON l.COUNTRY_ID = c.COUNTRY_ID
WHERE l.CITY <> 'Seattle' ORDER BY d.LOCATION_ID DESC;

-- Display view named VIEW_DEPT_LOCATE_COUNTRY
SELECT * FROM VIEW_DEPT_LOCATE_COUNTRY;
