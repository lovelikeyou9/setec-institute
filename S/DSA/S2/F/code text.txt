const express = require('express');
const mongoose = require('mongoose');
const bodyParser = require('body-parser');
const Appointments = require('./models/appointments');

const app = express();
const PORT = 3000;

const path = require('path');
app.use(express.static(path.join(__dirname, 'Public')));

app.use(express.urlencoded({ extended: true }));


app.use(bodyParser.json());
mongoose.connect('mongodb://localhost:27017/hospital', {
  useNewUrlParser: true,
  useUnifiedTopology: true,
});
const db = mongoose.connection;
db.on('error', console.error.bind(console, 'MongoDB connection error:'));
db.once('open', () => console.log('Connected to MongoDB locally'));






// Add to appointments
app.post('/submit', async (req, res) => {
  try {
    // Use fields according to your schema (not cat, supplier, tel)
    const { appointment_id, patient_id, doctor_id, department, appointment_date, duration, status, reason, notes } = req.body;

    // Create a new appointment document
    await Appointments.create({ appointment_id, patient_id, doctor_id, department, appointment_date, duration, status, reason, notes });

    res.redirect('/appointments.html'); // or wherever you want to go after submitting
  } catch (error) {
    res.status(500).send('Error saving appointment: ' + error.message);
  }
});

// delete appointment
app.post('/delete/:id', async (req, res) => {
  try {
    await Appointments.findByIdAndDelete(req.params.id);
    res.redirect('/appointments.html');
  } catch (error) {
    res.status(500).send('Error deleting appointment: ' + error.message);
  }
});

const mongoose = require('mongoose');

const appointmentsSchema = new mongoose.Schema({
    appointment_id: { type: String, required: true, trim: true },
    patient_id: { type: String, required: true, trim: true },
    doctor_id: { type: String, required: true, trim: true },
    department: { type: String, required: true, trim: true },
    appointment_date: { type: Date, required: true },
    duration: { type: Number, required: true },
    status: { type: String, required: true, trim: true },
    reason: { type: String, trim: true },
    notes: { type: String, trim: true },
    created_at: { type: Date, default: Date.now }
}, {
    versionKey: false
});

module.exports = mongoose.model('Appointments', appointmentsSchema);
